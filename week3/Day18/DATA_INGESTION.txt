Type: Object           <============= VERY IMPORTANT (else the json will be stringified!)
Expression:
{
"fileExtension": "{{ $json.fileExtension }}",
"mimeType": "{{ $json.mimeType }}",
"fileName": "{{ $json.fileName }}",
"fileSize": "{{ $json.fileSize }}",
"numpages": "{{ $json.numpages }}",

"PDFFormatVersion": "{{ $json.info.PDFFormatVersion }}",

"Author": "{{ $json.info.Author }}",
"CreationDate": "{{ $json.info.CreationDate }}",
"Title": "{{ $json.info.Title }}"
}
